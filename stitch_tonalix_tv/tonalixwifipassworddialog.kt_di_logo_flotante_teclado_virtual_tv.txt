/**
 * Diálogo modal flotante optimizado para televisores y mandos D-Pad.
 * Permite introducir contraseñas WPA2/WPA3 mediante un teclado visual sin requerir
 * teclado físico o el engorroso IME estándar de Android TV.
 *
 * @param accessPoint Punto de acceso seleccionado al que se desea conectar.
 * @param isConnecting Bandera que indica si la negociación del handshake está en curso.
 * @param onDismissRequest Callback al pulsar ATRÁS o cerrar el diálogo.
 * @param onConnect Callback ejecutado con la contraseña ingresada y opción de recordar red.
 */
@Composable
fun TonalixWifiPasswordDialog(
    accessPoint: WifiAccessPoint,
    isConnecting: Boolean = false,
    errorMessage: String? = null,
    onDismissRequest: () -> Unit,
    onConnect: (password: String, saveNetwork: Boolean) -> Unit
) {
    var passwordText by remember { mutableStateOf("") }
    var isPasswordVisible by remember { mutableStateOf(false) }
    var saveNetworkChecked by remember { mutableStateOf(true) }
    var keyboardMode by remember { mutableStateOf(KeyboardLayoutMode.LOWERCASE) }

    Dialog(
        onDismissRequest = onDismissRequest,
        properties = DialogProperties(
            usePlatformDefaultWidth = false,
            dismissOnBackPress = !isConnecting,
            dismissOnClickOutside = false
        )
    ) {
        Box(
            modifier = Modifier
                .fillMaxSize()
                .background(TonalixDialogColors.Scrim),
            contentAlignment = Alignment.Center
        ) {
            // Tarjeta principal del diálogo flotante
            Box(
                modifier = Modifier
                    .width(880.dp)
                    .clip(RoundedCornerShape(16.dp))
                    .background(TonalixDialogColors.DialogBackground)
                    .border(
                        width = 1.dp,
                        brush = Brush.verticalGradient(
                            listOf(
                                TonalixDialogColors.PrimaryCyan.copy(alpha = 0.6f),
                                TonalixDialogColors.BorderSubtle
                            )
                        ),
                        shape = RoundedCornerShape(16.dp)
                    )
                    .shadow(elevation = 32.dp, shape = RoundedCornerShape(16.dp))
                    .padding(26.dp)
            ) {
                Column(
                    modifier = Modifier.fillMaxWidth(),
                    verticalArrangement = Arrangement.spacedBy(16.dp)
                ) {
                    // 1. Cabecera con datos del AP y badge de seguridad
                    WifiDialogHeader(
                        accessPoint = accessPoint,
                        onDismiss = onDismissRequest,
                        isConnecting = isConnecting
                    )

                    Divider(color = TonalixDialogColors.BorderSubtle, thickness = 1.dp)

                    // 2. Campo de contraseña visible/oculta + visor interactivo
                    WifiPasswordFieldDisplay(
                        password = passwordText,
                        isVisible = isPasswordVisible,
                        onToggleVisibility = { isPasswordVisible = !isPasswordVisible },
                        onClearText = { passwordText = "" },
                        errorMessage = errorMessage,
                        isConnecting = isConnecting
                    )

                    // 3. Teclado Virtual en Pantalla (Diseño QWERTY TV en Grid)
                    WifiVirtualKeyboard(
                        currentMode = keyboardMode,
                        onKeyPress = { char ->
                            if (passwordText.length < 64) {
                                passwordText += char
                            }
                        },
                        onBackspace = {
                            if (passwordText.isNotEmpty()) {
                                passwordText = passwordText.dropLast(1)
                            }
                        },
                        onToggleMode = { newMode ->
                            keyboardMode = newMode
                        },
                        onSpace = {
                            if (passwordText.length < 64) passwordText += " "
                        },
                        isConnecting = isConnecting
                    )

                    Divider(color = TonalixDialogColors.BorderSubtle, thickness = 1.dp)

                    // 4. Barra Inferior de Acciones y Guía D-Pad
                    WifiDialogBottomActions(
                        passwordLength = passwordText.length,
                        saveNetwork = saveNetworkChecked,
                        onToggleSaveNetwork = { saveNetworkChecked = it },
                        isConnecting = isConnecting,
                        onCancel = onDismissRequest,
                        onConnect = {
                            onConnect(passwordText, saveNetworkChecked)
                        }
                    )
                }
            }
        }
    }
}

@Composable
private fun WifiPasswordFieldDisplay(
    password: String,
    isVisible: Boolean,
    onToggleVisibility: () -> Unit,
    onClearText: () -> Unit,
    errorMessage: String?,
    isConnecting: Boolean
) {
    Column {
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .clip(RoundedCornerShape(10.dp))
                .background(TonalixDialogColors.DialogSurface)
                .border(
                    width = 1.5.dp,
                    color = if (errorMessage != null) TonalixDialogColors.DangerRed else TonalixDialogColors.BorderSubtle,
                    shape = RoundedCornerShape(10.dp)
                )
                .padding(horizontal = 16.dp, vertical = 12.dp)
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                Row(verticalAlignment = Alignment.CenterVertically, modifier = Modifier.weight(1f)) {
                    Icon(
                        imageVector = Icons.Default.Key,
                        contentDescription = null,
                        tint = if (password.isNotEmpty()) TonalixDialogColors.PrimaryCyan else TonalixDialogColors.TextMuted,
                        modifier = Modifier.size(20.dp)
                    )
                    Spacer(modifier = Modifier.width(12.dp))

                    if (password.isEmpty()) {
                        Text(
                            text = "Introduce la clave de red (mínimo 8 caracteres)...",
                            color = TonalixDialogColors.TextMuted,
                            fontSize = 14.sp
                        )
                    } else {
                        Text(
                            text = if (isVisible) password else "•".repeat(password.length),
                            color = TonalixDialogColors.TextPrimary,
                            fontSize = 16.sp,
                            fontWeight = FontWeight.Bold,
                            fontFamily = FontFamily.Monospace,
                            letterSpacing = if (isVisible) 1.sp else 3.sp,
                            maxLines = 1,
                            overflow = TextOverflow.Ellipsis
                        )
                    }
                }

                Row(verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                    // Contador de caracteres
                    Text(
                        text = "${password.length}/64",
                        color = if (password.length >= 8) TonalixDialogColors.AccentEmerald else TonalixDialogColors.TextMuted,
                        fontSize = 11.sp,
                        fontFamily = FontFamily.Monospace,
                        fontWeight = FontWeight.Bold
                    )

                    // Toggle Ver Contraseña
                    Box(
                        modifier = Modifier
                            .clip(RoundedCornerShape(6.dp))
                            .background(TonalixDialogColors.KeySurface)
                            .border(1.dp, TonalixDialogColors.BorderSubtle, RoundedCornerShape(6.dp))
                            .clickable(enabled = !isConnecting, onClick = onToggleVisibility)
                            .padding(horizontal = 8.dp, vertical = 6.dp)
                    ) {
                        Row(verticalAlignment = Alignment.CenterVertically) {
                            Icon(
                                imageVector = if (isVisible) Icons.Default.VisibilityOff else Icons.Default.Visibility,
                                contentDescription = null,
                                tint = if (isVisible) TonalixDialogColors.PrimaryCyan else TonalixDialogColors.TextSecondary,
                                modifier = Modifier.size(15.dp)
                            )
                            Spacer(modifier = Modifier.width(4.dp))
                            Text(
                                text = if (isVisible) "Ocultar" else "Mostrar",
                                color = if (isVisible) TonalixDialogColors.PrimaryCyan else TonalixDialogColors.TextSecondary,
                                fontSize = 10.sp,
                                fontWeight = FontWeight.SemiBold
                            )
                        }
                    }

                    // Botón Borrar Todo
                    if (password.isNotEmpty()) {
                        Box(
                            modifier = Modifier
                                .clip(RoundedCornerShape(6.dp))
                                .background(TonalixDialogColors.KeySurface)
                                .border(1.dp, TonalixDialogColors.BorderSubtle, RoundedCornerShape(6.dp))
                                .clickable(enabled = !isConnecting, onClick = onClearText)
                                .padding(horizontal = 8.dp, vertical = 6.dp)
                        ) {
                            Row(verticalAlignment = Alignment.CenterVertically) {
                                Icon(
                                    imageVector = Icons.Default.Clear,
                                    contentDescription = null,
                                    tint = TonalixDialogColors.DangerRed,
                                    modifier = Modifier.size(15.dp)
                                )
                                Spacer(modifier = Modifier.width(4.dp))
                                Text(
                                    text = "Limpiar",
                                    color = TonalixDialogColors.DangerRed,
                                    fontSize = 10.sp,
                                    fontWeight = FontWeight.SemiBold
                                )
                            }
                        }
                    }
                }
            }
        }

        // Mensaje de advertencia o error si existe
        if (errorMessage != null) {
            Spacer(modifier = Modifier.height(4.dp))
            Row(verticalAlignment = Alignment.CenterVertically, modifier = Modifier.padding(horizontal = 4.dp)) {
                Icon(
                    imageVector = Icons.Default.Warning,
                    contentDescription = null,
                    tint = TonalixDialogColors.DangerRed,
                    modifier = Modifier.size(13.dp)
                )
                Spacer(modifier = Modifier.width(6.dp))
                Text(
                    text = errorMessage,
                    color = TonalixDialogColors.DangerRed,
                    fontSize = 11.sp,
                    fontWeight = FontWeight.Medium
                )
            }
        }
    }
}

// ============================================================================
// TECLADO VIRTUAL EN PANTALLA (COMPOSABLE COMPACTO PARA TV / D-PAD)
// ============================================================================
@Composable
private fun WifiVirtualKeyboard(
    currentMode: KeyboardLayoutMode,
    onKeyPress: (String) -> Unit,
    onBackspace: () -> Unit,
    onToggleMode: (KeyboardLayoutMode) -> Unit,
    onSpace: () -> Unit,
    isConnecting: Boolean
) {
    // Definición de filas de teclas según el modo
    val row1 = when (currentMode) {
        KeyboardLayoutMode.LOWERCASE -> listOf("1", "2", "3", "4", "5", "6", "7", "8", "9", "0")
        KeyboardLayoutMode.UPPERCASE -> listOf("1", "2", "3", "4", "5", "6", "7", "8", "9", "0")
        KeyboardLayoutMode.SYMBOLS_NUMBERS -> listOf("!", "@", "#", "$", "%", "^", "&", "*", "(", ")")
    }

    val row2 = when (currentMode) {
        KeyboardLayoutMode.LOWERCASE -> listOf("q", "w", "e", "r", "t", "y", "u", "i", "o", "p")
        KeyboardLayoutMode.UPPERCASE -> listOf("Q", "W", "E", "R", "T", "Y", "U", "I", "O", "P")
        KeyboardLayoutMode.SYMBOLS_NUMBERS -> listOf("-", "_", "=", "+", "[", "]", "{", "}", "\\", "|")
    }

    val row3 = when (currentMode) {
        KeyboardLayoutMode.LOWERCASE -> listOf("a", "s", "d", "f", "g", "h", "j", "k", "l", "ñ")
        KeyboardLayoutMode.UPPERCASE -> listOf("A", "S", "D", "F", "G", "H", "J", "K", "L", "Ñ")
        KeyboardLayoutMode.SYMBOLS_NUMBERS -> listOf(";", ":", "'", "\"", ",", ".", "<", ">", "/", "?")
    }

    val row4 = when (currentMode) {
        KeyboardLayoutMode.LOWERCASE -> listOf("z", "x", "c", "v", "b", "n", "m", "-", "_", ".")
        KeyboardLayoutMode.UPPERCASE -> listOf("Z", "X", "C", "V", "B", "N", "M", "-", "_", ".")
        KeyboardLayoutMode.SYMBOLS_NUMBERS -> listOf("~", "`", "€", "$", "£", "¥", "§", "°", "«", "»")
    }

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(TonalixDialogColors.DialogSurface)
            .border(1.dp, TonalixDialogColors.BorderSubtle, RoundedCornerShape(12.dp))
            .padding(14.dp),
        verticalArrangement = Arrangement.spacedBy(8.dp)
    ) {
        // Fila 1: Números / Símbolos superiores
        KeyboardRow(keys = row1, onKeyPress = onKeyPress, isEnabled = !isConnecting)

        // Fila 2: Letras superiores
        KeyboardRow(keys = row2, onKeyPress = onKeyPress, isEnabled = !isConnecting)

        // Fila 3: Letras intermedias
        KeyboardRow(keys = row3, onKeyPress = onKeyPress, isEnabled = !isConnecting)

        // Fila 4: Letras inferiores y caracteres frecuentes
        KeyboardRow(keys = row4, onKeyPress = onKeyPress, isEnabled = !isConnecting)

        // Fila 5: Teclas de Control y Modificadores (Shift, ?123, Espacio, Backspace)
        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.spacedBy(8.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            // Tecla Mayúsculas / Minúsculas (Shift)
            KeyboardSpecialKey(
                label = if (currentMode == KeyboardLayoutMode.UPPERCASE) "MAYÚS [ON]" else "Mayús",
                icon = Icons.Default.KeyboardArrowUp,
                isActive = currentMode == KeyboardLayoutMode.UPPERCASE,
                modifier = Modifier.weight(1.8f),
                isEnabled = !isConnecting,
                onClick = {
                    onToggleMode(
                        if (currentMode == KeyboardLayoutMode.UPPERCASE) KeyboardLayoutMode.LOWERCASE else KeyboardLayoutMode.UPPERCASE
                    )
                }
            )

            // Tecla Símbolos / Números (?123)
            KeyboardSpecialKey(
                label = if (currentMode == KeyboardLayoutMode.SYMBOLS_NUMBERS) "ABC" else "?123",
                icon = Icons.Default.Pin,
                isActive = currentMode == KeyboardLayoutMode.SYMBOLS_NUMBERS,
                modifier = Modifier.weight(1.5f),
                isEnabled = !isConnecting,
                onClick = {
                    onToggleMode(
                        if (currentMode == KeyboardLayoutMode.SYMBOLS_NUMBERS) KeyboardLayoutMode.LOWERCASE else KeyboardLayoutMode.SYMBOLS_NUMBERS
                    )
                }
            )

            // Barra de Espacio
            KeyboardSpecialKey(
                label = "Espacio",
                icon = Icons.Default.SpaceBar,
                isActive = false,
                modifier = Modifier.weight(3.5f),
                isEnabled = !isConnecting,
                onClick = onSpace
            )

            // Tecla Borrar (Backspace)
            KeyboardSpecialKey(
                label = "Borrar",
                icon = Icons.Default.Backspace,
                isActive = false,
                modifier = Modifier.weight(1.8f),
                isEnabled = !isConnecting,
                onClick = onBackspace
            )
        }
    }
}

@Composable
private fun KeyboardRow(
    keys: List<String>,
    onKeyPress: (String) -> Unit,
    isEnabled: Boolean
) {
    Row(
        modifier = Modifier.fillMaxWidth(),
        horizontalArrangement = Arrangement.spacedBy(6.dp)
    ) {
        keys.forEach { key ->
            KeyboardCharacterKey(
                character = key,
                modifier = Modifier.weight(1f),
                isEnabled = isEnabled,
                onClick = { onKeyPress(key) }
            )
        }
    }
}

@Composable
private fun KeyboardCharacterKey(
    character: String,
    modifier: Modifier = Modifier,
    isEnabled: Boolean,
    onClick: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val bgColor by animateColorAsState(
        targetValue = when {
            isFocused -> TonalixDialogColors.PrimaryCyan
            else -> TonalixDialogColors.KeySurface
        },
        label = "keyBg"
    )

    val textColor by animateColorAsState(
        targetValue = when {
            isFocused -> Color.Black
            else -> TonalixDialogColors.TextPrimary
        },
        label = "keyText"
    )

    Box(
        modifier = modifier
            .height(44.dp)
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(
                width = 1.dp,
                color = if (isFocused) TonalixDialogColors.PrimaryCyan else TonalixDialogColors.BorderSubtle,
                shape = RoundedCornerShape(8.dp)
            )
            .focusable(enabled = isEnabled, interactionSource = interactionSource)
            .clickable(enabled = isEnabled, onClick = onClick),
        contentAlignment = Alignment.Center
    ) {
        Text(
            text = character,
            color = textColor,
            fontSize = 15.sp,
            fontWeight = if (isFocused) FontWeight.Bold else FontWeight.SemiBold,
            fontFamily = FontFamily.Monospace,
            textAlign = TextAlign.Center
        )
    }
}

@Composable
private fun KeyboardSpecialKey(
    label: String,
    icon: ImageVector,
    isActive: Boolean,
    modifier: Modifier = Modifier,
    isEnabled: Boolean,
    onClick: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val bgColor by animateColorAsState(
        targetValue = when {
            isFocused -> TonalixDialogColors.PrimaryCyan
            isActive -> TonalixDialogColors.KeySurfaceActive
            else -> TonalixDialogColors.KeySurface
        },
        label = "specialKeyBg"
    )

    val contentColor by animateColorAsState(
        targetValue = when {
            isFocused -> Color.Black
            isActive -> TonalixDialogColors.PrimaryCyan
            else -> TonalixDialogColors.TextSecondary
        },
        label = "specialKeyContent"
    )

    Box(
        modifier = modifier
            .height(44.dp)
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(
                width = 1.dp,
                color = when {
                    isFocused -> TonalixDialogColors.PrimaryCyan
                    isActive -> TonalixDialogColors.PrimaryCyan.copy(alpha = 0.5f)
                    else -> TonalixDialogColors.BorderSubtle
                },
                shape = RoundedCornerShape(8.dp)
            )
            .focusable(enabled = isEnabled, interactionSource = interactionSource)
            .clickable(enabled = isEnabled, onClick = onClick)
            .padding(horizontal = 8.dp),
        contentAlignment = Alignment.Center
    ) {
        Row(
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.Center
        ) {
            Icon(
                imageVector = icon,
                contentDescription = null,
                tint = contentColor,
                modifier = Modifier.size(16.dp)
            )
            Spacer(modifier = Modifier.width(6.dp))
            Text(
                text = label,
                color = contentColor,
                fontSize = 12.sp,
                fontWeight = FontWeight.SemiBold,
                fontFamily = FontFamily.SansSerif
            )
        }
    }
}

// ============================================================================
// ACCIONES INFERIORES & TOGGLE RECORDAR RED
// ============================================================================
@Composable
private fun WifiDialogBottomActions(
    passwordLength: Int,
    saveNetwork: Boolean,
    onToggleSaveNetwork: (Boolean) -> Unit,
    isConnecting: Boolean,
    onCancel: () -> Unit,
    onConnect: () -> Unit
) {
    val canConnect = passwordLength >= 8 && !isConnecting

    Row(
        modifier = Modifier.fillMaxWidth(),
        horizontalArrangement = Arrangement.SpaceBetween,
        verticalAlignment = Alignment.CenterVertically
    ) {
        // Toggle Recordar Red y Clave
        Row(
            verticalAlignment = Alignment.CenterVertically,
            modifier = Modifier
                .clip(RoundedCornerShape(6.dp))
                .clickable(enabled = !isConnecting) { onToggleSaveNetwork(!saveNetwork) }
                .padding(horizontal = 6.dp, vertical = 4.dp)
        ) {
            Switch(
                checked = saveNetwork,
                onCheckedChange = onToggleSaveNetwork,
                enabled = !isConnecting,
                colors = SwitchDefaults.colors(
                    checkedThumbColor = Color.Black,
                    checkedTrackColor = TonalixDialogColors.PrimaryCyan,
                    uncheckedThumbColor = TonalixDialogColors.TextSecondary,
                    uncheckedTrackColor = TonalixDialogColors.KeySurface
                ),
                modifier = Modifier.scale(0.85f)
            )
            Spacer(modifier = Modifier.width(8.dp))
            Column {
                Text(
                    text = "Recordar esta red",
                    color = TonalixDialogColors.TextPrimary,
                    fontSize = 12.sp,
                    fontWeight = FontWeight.SemiBold
                )
                Text(
                    text = "Conectar automáticamente al encender la TV",
                    color = TonalixDialogColors.TextMuted,
                    fontSize = 9.sp
                )
            }
        }

        // Botones Cancelar y Conectar
        Row(horizontalArrangement = Arrangement.spacedBy(10.dp), verticalAlignment = Alignment.CenterVertically) {
            OutlinedButton(
                onClick = onCancel,
                enabled = !isConnecting,
                colors = ButtonDefaults.outlinedButtonColors(containerColor = TonalixDialogColors.KeySurface),
                border = androidx.compose.foundation.BorderStroke(1.dp, TonalixDialogColors.BorderSubtle),
                shape = RoundedCornerShape(8.dp),
                contentPadding = PaddingValues(horizontal = 18.dp, vertical = 10.dp)
            ) {
                Text(
                    text = "Cancelar",
                    color = TonalixDialogColors.TextSecondary,
                    fontSize = 12.sp,
                    fontWeight = FontWeight.SemiBold
                )
            }

            Button(
                onClick = onConnect,
                enabled = canConnect,
                colors = ButtonDefaults.buttonColors(
                    containerColor = TonalixDialogColors.PrimaryCyan,
                    disabledContainerColor = TonalixDialogColors.KeySurface
                ),
                shape = RoundedCornerShape(8.dp),
                contentPadding = PaddingValues(horizontal = 22.dp, vertical = 10.dp)
            ) {
                if (isConnecting) {
                    CircularProgressIndicator(
                        modifier = Modifier.size(16.dp),
                        color = Color.Black,
                        strokeWidth = 2.dp
                    )
                    Spacer(modifier = Modifier.width(8.dp))
                    Text(
                        text = "Negociando WPA3...",
                        color = Color.Black,
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Bold
                    )
                } else {
                    Icon(
                        imageVector = Icons.Default.Wifi,
                        contentDescription = null,
                        tint = if (canConnect) Color.Black else TonalixDialogColors.TextMuted,
                        modifier = Modifier.size(16.dp)
                    )
                    Spacer(modifier = Modifier.width(8.dp))
                    Text(
                        text = "Conectar [OK]",
                        color = if (canConnect) Color.Black else TonalixDialogColors.TextMuted,
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Bold
                    )
                }
            }
        }
    }
}
