/**
 * Pantalla principal de Acerca de y Actualizaciones del Sistema
 * Optimizada para control remoto D-Pad / Smart TV y navegación fluida.
 */
@Composable
fun TonalixAboutScreen(
    onNavigateBack: () -> Unit = {},
    onCheckForUpdates: () -> Unit = {},
    onRestartSystem: () -> Unit = {},
    onOpenDeveloperOptions: () -> Unit = {},
    onOpenChangelog: () -> Unit = {},
    onSendDiagnosticReport: () -> Unit = {},
    onFactoryReset: () -> Unit = {},
    onRunHardwareDiagnostics: () -> Unit = {}
) {
    var selectedCategory by remember { mutableStateOf(SettingsCategory.ABOUT) }
    var selectedChannel by remember { mutableStateOf(UpdateChannel.STABLE) }
    var isCheckingUpdates by remember { mutableStateOf(false) }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(TonalixColors.Background)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            // Barra superior fija con estado del sistema y perfil blindado
            TonalixTopBar()

            // Cuerpo principal: Menú lateral + Contenido central + Panel de telemetría e integridad
            Row(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
            ) {
                // Sidebar de categorías
                TonalixSettingsSidebar(
                    selectedCategory = selectedCategory,
                    onSelectCategory = { selectedCategory = it }
                )

                // Separador vertical sutil
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // Contenido principal de Acerca de
                Box(
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                ) {
                    LazyColumn(
                        modifier = Modifier
                            .fillMaxSize()
                            .padding(horizontal = 28.dp, vertical = 20.dp),
                        verticalArrangement = Arrangement.spacedBy(20.dp)
                    ) {
                        item {
                            AboutHeaderSection(
                                isCheckingUpdates = isCheckingUpdates,
                                onCheckForUpdates = {
                                    isCheckingUpdates = true
                                    onCheckForUpdates()
                                },
                                onRestartSystem = onRestartSystem
                            )
                        }

                        item {
                            CompilationAndChannelSection(
                                selectedChannel = selectedChannel,
                                onSelectChannel = { selectedChannel = it }
                            )
                        }

                        item {
                            InstalledHighlightsSection(onOpenChangelog = onOpenChangelog)
                        }

                        item {
                            HardwareAndCertificationsRow()
                        }

                        item {
                            MaintenanceAndDebuggingSection(
                                onOpenDeveloperOptions = onOpenDeveloperOptions,
                                onOpenChangelog = onOpenChangelog,
                                onSendDiagnosticReport = onSendDiagnosticReport,
                                onFactoryReset = onFactoryReset
                            )
                        }
                    }
                }

                // Separador vertical
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // Panel lateral de Integridad del Sistema y Telemetría
                TonalixIntegritySidebar(
                    onRunHardwareDiagnostics = onRunHardwareDiagnostics
                )
            }

            // Barra inferior con guía de navegación D-PAD y estado del sistema
            TonalixBottomBar()
        }
    }
}

@Composable
private fun SidebarCategoryItem(
    category: SettingsCategory,
    isSelected: Boolean,
    onClick: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val bgColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyanGlow
            isFocused -> TonalixColors.SurfaceCardHover
            else -> Color.Transparent
        }, label = "bg"
    )

    val borderColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyan
            isFocused -> TonalixColors.TextSecondary
            else -> Color.Transparent
        }, label = "border"
    )

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(1.dp, borderColor, RoundedCornerShape(8.dp))
            .focusable(interactionSource = interactionSource)
            .clickable(onClick = onClick)
            .padding(horizontal = 12.dp, vertical = 10.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Icon(
            imageVector = category.icon,
            contentDescription = null,
            tint = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextSecondary,
            modifier = Modifier.size(18.dp)
        )
        Spacer(modifier = Modifier.width(10.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = category.title,
                color = if (isSelected) TonalixColors.TextPrimary else TonalixColors.TextSecondary,
                fontSize = 12.sp,
                fontWeight = if (isSelected) FontWeight.SemiBold else FontWeight.Normal
            )
            Text(
                text = category.subtitle,
                color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextMuted,
                fontSize = 10.sp,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
        }
        if (isSelected) {
            Box(
                modifier = Modifier
                    .size(6.dp)
                    .clip(CircleShape)
                    .background(TonalixColors.PrimaryCyan)
            )
        }
    }
}
