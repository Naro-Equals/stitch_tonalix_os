/**
 * Pantalla de Calibración de Pantalla y Salida de Video para Tonalix TV OS.
 * Optimizada para control remoto D-Pad 10-foot UI en Smart TV / Android TV Boxes.
 */
@Composable
fun TonalixDisplayScreen(
    onNavigateBack: () -> Unit = {},
    onResetValues: () -> Unit = {},
    onApplySettings: () -> Unit = {},
    onRunStressBenchmark: () -> Unit = {},
    onRecalibrateRoomAcoustics: () -> Unit = {}
) {
    var selectedCategory by remember { mutableStateOf(SettingsCategory.DISPLAY) }
    var selectedResolution by remember { mutableStateOf(ResolutionProfile.RECOMMENDED_4K) }
    var selectedAppProfile by remember { mutableStateOf(AppPictureProfile.PLEX_CINE) }

    // Toggles de calibración de imagen y video
    var afrEnabled by remember { mutableStateOf(true) }
    var vrrEnabled by remember { mutableStateOf(true) }
    var dolbyVisionEnabled by remember { mutableStateOf(true) }
    var filmmakerModeEnabled by remember { mutableStateOf(true) }
    var allmGamingEnabled by remember { mutableStateOf(true) }
    var vocalBoostEnabled by remember { mutableStateOf(true) }
    var nightCompressionEnabled by remember { mutableStateOf(true) }
    var superResolutionEnabled by remember { mutableStateOf(true) }

    // Sliders de calibración fina
    var hdrPeakBrightness by remember { mutableFloatStateOf(850f) }
    var memcLevel by remember { mutableStateOf("Film Purist (Nivel 2)") }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(TonalixColors.Background)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            // Barra superior fija con estado del sistema y telemetría
            TonalixTopBar()

            // Cuerpo principal: Sidebar Izquierdo + Contenido Central + Sidebar Derecho (Telemetría y GPU)
            Row(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
            ) {
                // 1. Sidebar de navegación lateral
                TonalixSettingsSidebar(
                    selectedCategory = selectedCategory,
                    onSelectCategory = { selectedCategory = it }
                )

                // Separador vertical sutil
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 2. Contenido Central (Calibración Pantalla y Video)
                Box(
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                ) {
                    LazyColumn(
                        modifier = Modifier
                            .fillMaxSize()
                            .padding(horizontal = 26.dp, vertical = 20.dp),
                        verticalArrangement = Arrangement.spacedBy(18.dp)
                    ) {
                        // Encabezado y acciones principales
                        item {
                            DisplayHeaderSection(
                                onResetValues = onResetValues,
                                onApplySettings = onApplySettings
                            )
                        }

                        // Banner de Detección Automática de Hardware y Panel HDMI 2.1
                        item {
                            HardwareDetectionBanner()
                        }

                        // Perfiles de Configuración Rápida por App / Escena
                        item {
                            PictureProfilesSection(
                                selectedProfile = selectedAppProfile,
                                onSelectProfile = { selectedAppProfile = it }
                            )
                        }

                        // Tarjeta Principal: Resolución, Tasa de Refresco, AFR y VRR
                        item {
                            ResolutionAndRefreshCard(
                                selectedResolution = selectedResolution,
                                onSelectResolution = { selectedResolution = it },
                                afrEnabled = afrEnabled,
                                onToggleAfr = { afrEnabled = it },
                                vrrEnabled = vrrEnabled,
                                onToggleVrr = { vrrEnabled = it }
                            )
                        }

                        // Tarjeta Super-Resolución y Nitidez Neuronal AI
                        item {
                            SuperResolutionCard(
                                isEnabled = superResolutionEnabled,
                                onToggle = { superResolutionEnabled = it }
                            )
                        }

                        // Compensación de Movimiento MEMC y Baja Latencia ALLM
                        item {
                            MemcAndLatencyCard(
                                selectedMemc = memcLevel,
                                onSelectMemc = { memcLevel = it },
                                allmEnabled = allmGamingEnabled,
                                onToggleAllm = { allmGamingEnabled = it }
                            )
                        }

                        // Tarjeta: Espacio de Color, Dolby Vision, Filmmaker Mode y Brillo HDR
                        item {
                            ColorSpaceAndHdrCard(
                                dolbyVision = dolbyVisionEnabled,
                                onToggleDolbyVision = { dolbyVisionEnabled = it },
                                filmmaker = filmmakerModeEnabled,
                                onToggleFilmmaker = { filmmakerModeEnabled = it },
                                peakBrightness = hdrPeakBrightness,
                                onBrightnessChange = { hdrPeakBrightness = it }
                            )
                        }

                        // Ecualizador Dinámico de Diálogos y Sala (Audio HDMI eARC Passthrough)
                        item {
                            AudioCalibrationSection(
                                vocalBoost = vocalBoostEnabled,
                                onToggleVocalBoost = { vocalBoostEnabled = it },
                                nightCompression = nightCompressionEnabled,
                                onToggleNightCompression = { nightCompressionEnabled = it },
                                onRecalibrateRoomAcoustics = onRecalibrateRoomAcoustics
                            )
                        }
                    }
                }

                // Separador vertical
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 3. Panel Lateral Derecho (Diagnóstico en Vivo y Ficha de Decodificación)
                TonalixDisplayDiagnosticsSidebar(
                    onRunBenchmark = onRunStressBenchmark
                )
            }

            // Barra inferior con guía de botones de control remoto D-PAD
            TonalixBottomBar()
        }
    }
}

@Composable
private fun SidebarCategoryItem(
    category: SettingsCategory,
    isSelected: Boolean,
    onClick: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val bgColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyanGlow
            isFocused -> TonalixColors.SurfaceCardHover
            else -> Color.Transparent
        }, label = "bg"
    )

    val borderColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyan
            isFocused -> TonalixColors.TextSecondary
            else -> Color.Transparent
        }, label = "border"
    )

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(1.dp, borderColor, RoundedCornerShape(8.dp))
            .focusable(interactionSource = interactionSource)
            .clickable(onClick = onClick)
            .padding(horizontal = 12.dp, vertical = 10.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Icon(
            imageVector = category.icon,
            contentDescription = null,
            tint = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextSecondary,
            modifier = Modifier.size(18.dp)
        )
        Spacer(modifier = Modifier.width(10.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = category.title,
                color = if (isSelected) TonalixColors.TextPrimary else TonalixColors.TextSecondary,
                fontSize = 12.sp,
                fontWeight = if (isSelected) FontWeight.SemiBold else FontWeight.Normal
            )
            Text(
                text = category.subtitle,
                color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextMuted,
                fontSize = 10.sp,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
        }
        if (isSelected) {
            Box(
                modifier = Modifier
                    .size(6.dp)
                    .clip(CircleShape)
                    .background(TonalixColors.PrimaryCyan)
            )
        }
    }
}
