/**
 * Punto de entrada con recolección de ciclo de vida consciente para Android TV / Compose TV.
 */
@Composable
fun TonalixNetworkRoute(
    viewModel: TonalixNetworkViewModel = viewModel(),
    onNavigateBack: () -> Unit = {}
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is NetworkUiEffect.ShowToast -> {
                    Toast.makeText(context, effect.message, Toast.LENGTH_SHORT).show()
                }
                is NetworkUiEffect.PlaySound -> {
                    // Integración con SoundPool / View.playSoundEffect del control remoto
                }
                is NetworkUiEffect.NavigateBack -> {
                    onNavigateBack()
                }
            }
        }
    }

    TonalixNetworkScreen(
        uiState = uiState,
        onAction = viewModel::onAction,
        onNavigateBack = onNavigateBack
    )
}

/**
 * Pantalla pura / desacoplada para previews y testing sin dependencias duras de Android.
 */
@Composable
fun TonalixNetworkScreen(
    uiState: NetworkUiState,
    onAction: (NetworkUserAction) -> Unit,
    onNavigateBack: () -> Unit = {}
) {
    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(TonalixColors.Background)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            // Barra superior fija con estado del sistema y telemetría de red
            TonalixTopBar(telemetry = uiState.telemetry)

            // Cuerpo principal: Sidebar Izquierdo + Contenido Central + Sidebar Derecho (Diagnóstico IP y Enrutamiento)
            Row(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
            ) {
                // 1. Sidebar de navegación lateral
                TonalixSettingsSidebar(
                    selectedCategory = SettingsCategory.NETWORK,
                    onSelectCategory = { /* Navegación de categorías */ }
                )

                // Separador vertical sutil
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 2. Contenido Central (Wi-Fi 6, APs Cercanos, Mesh Roaming y Filtros)
                Box(
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                ) {
                    LazyColumn(
                        modifier = Modifier
                            .fillMaxSize()
                            .padding(horizontal = 24.dp, vertical = 18.dp),
                        verticalArrangement = Arrangement.spacedBy(16.dp)
                    ) {
                        // Encabezado y acciones principales (Escanear Redes y Test de Ping)
                        item {
                            NetworkHeaderSection(
                                isScanning = uiState.isScanningNetworks,
                                isTesting = uiState.isRunningSpeedPingTest,
                                onScanNetworks = { onAction(NetworkUserAction.ScanNetworks) },
                                onRunPingTest = { onAction(NetworkUserAction.RunSpeedAndPingTest) }
                            )
                        }

                        // Tarjeta de Enlace Primario Activo (Wi-Fi 6 / 5 GHz / WPA3-SAE)
                        item {
                            ActiveNetworkHeroCard(
                                activeNetwork = uiState.activeConnectedNetwork,
                                telemetry = uiState.telemetry,
                                onDisconnect = { ssid -> onAction(NetworkUserAction.DisconnectNetwork(ssid)) },
                                onForget = { ssid -> onAction(NetworkUserAction.ForgetNetwork(ssid)) }
                            )
                        }

                        // Tarjeta de Ajustes del Módulo Wi-Fi & Protocolos Avanzados (Mesh, QoS, DFS)
                        item {
                            WifiTogglesCard(
                                wifiEnabled = uiState.wifiHardwareEnabled,
                                onToggleWifi = { onAction(NetworkUserAction.ToggleWifiHardware(it)) },
                                meshRoaming = uiState.meshFastRoaming80211kvr,
                                onToggleMesh = { onAction(NetworkUserAction.ToggleMeshRoaming(it)) },
                                gamingQos = uiState.lowLatencyGamingQos,
                                onToggleQos = { onAction(NetworkUserAction.ToggleLowLatencyQos(it)) },
                                dfsChannels = uiState.dfsChannels5GhzAllowed,
                                onToggleDfs = { onAction(NetworkUserAction.ToggleDfsChannels(it)) }
                            )
                        }

                        // Pestañas de Filtro y Lista de Redes Wi-Fi Disponibles
                        item {
                            NearbyNetworksSection(
                                selectedTab = uiState.selectedFilterTab,
                                onSelectTab = { onAction(NetworkUserAction.SelectFilterTab(it)) },
                                nearbyNetworks = uiState.nearbyNetworks,
                                isConnecting = uiState.isConnectingNetwork,
                                connectingSsid = uiState.connectingSsid,
                                onConnect = { onAction(NetworkUserAction.ConnectToNetwork(it)) }
                            )
                        }
                    }
                }

                // Separador vertical
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 3. Panel Lateral Derecho (Parámetros IP, VLAN, DNS y Diagnóstico en Tiempo Real)
                TonalixNetworkDiagnosticsSidebar(
                    ipConfig = uiState.ipConfig,
                    telemetry = uiState.telemetry,
                    isRunningTest = uiState.isRunningSpeedPingTest,
                    testProgress = uiState.speedTestProgress,
                    onChangeIpMode = { onAction(NetworkUserAction.SetIpAssignmentMode(it)) },
                    onRunSpeedTest = { onAction(NetworkUserAction.RunSpeedAndPingTest) },
                    onFlushDns = { onAction(NetworkUserAction.FlushDnsAndResetRouting) }
                )
            }

            // Barra inferior con guía de botones de control remoto D-PAD
            TonalixBottomBar()
        }
    }
}

@Composable
private fun SidebarCategoryItem(
    category: SettingsCategory,
    isSelected: Boolean,
    onClick: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val bgColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyanGlow
            isFocused -> TonalixColors.SurfaceCardHover
            else -> Color.Transparent
        }, label = "bg"
    )

    val borderColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyan
            isFocused -> TonalixColors.TextSecondary
            else -> Color.Transparent
        }, label = "border"
    )

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(1.dp, borderColor, RoundedCornerShape(8.dp))
            .focusable(interactionSource = interactionSource)
            .clickable(onClick = onClick)
            .padding(horizontal = 12.dp, vertical = 10.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Icon(
            imageVector = category.icon,
            contentDescription = null,
            tint = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextSecondary,
            modifier = Modifier.size(18.dp)
        )
        Spacer(modifier = Modifier.width(10.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = category.title,
                color = if (isSelected) TonalixColors.TextPrimary else TonalixColors.TextSecondary,
                fontSize = 12.sp,
                fontWeight = if (isSelected) FontWeight.SemiBold else FontWeight.Normal
            )
            Text(
                text = category.subtitle,
                color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextMuted,
                fontSize = 10.sp,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
        }
        if (isSelected) {
            Box(
                modifier = Modifier
                    .size(6.dp)
                    .clip(CircleShape)
                    .background(TonalixColors.PrimaryCyan)
            )
        }
    }
}

    connectingSsid: String?,
    onConnect: (WifiAccessPoint) -> Unit

    ) {
        Column {
            Row(

                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Row(verticalAlignment = Alignment.CenterVertically) {

                        fontSize = 13.sp,
                        fontWeight = FontWeight.Bold
                    )
                }

                // Pestañas de filtrado de bandas
                Row(
                    modifier = Modifier

                        .border(1.dp, TonalixColors.SurfaceBorder, RoundedCornerShape(6.dp))
                        .padding(3.dp),
                    horizontalArrangement = Arrangement.spacedBy(4.dp)
                ) {
                    listOf("Todas", "5GHz / 6GHz", "2.4GHz", "Abiertas").forEachIndexed { index, tabTitle ->
                        val isSelected = selectedTab == index
                        Box(
                            modifier = Modifier
                                .clip(RoundedCornerShape(4.dp))
                                .background(if (isSelected) TonalixColors.PrimaryCyan else Color.Transparent)

                            Text(

@Composable
private fun WifiAccessPointRow(
    ap: WifiAccessPoint,
    isConnectingThis: Boolean,
    onConnect: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val borderColor = when {
        isFocused -> TonalixColors.PrimaryCyan
        else -> TonalixColors.SurfaceBorder
    }

    val bgColor = when {
        isFocused -> TonalixColors.SurfaceCardHover
        else -> Color(0xFF0A0F14)
    }

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(1.dp, borderColor, RoundedCornerShape(8.dp))
            .focusable(interactionSource = interactionSource)
            .clickable(onClick = onConnect)
            .padding(horizontal = 14.dp, vertical = 10.dp),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.SpaceBetween
    ) {
        Row(verticalAlignment = Alignment.CenterVertically, modifier = Modifier.weight(1f)) {
            Icon(
                imageVector = when {
                    ap.signalDbm > -60 -> Icons.Default.Wifi
                    ap.signalDbm > -75 -> Icons.Default.NetworkWifi2Bar
                    else -> Icons.Default.NetworkWifi1Bar
                },
                contentDescription = null,
                tint = if (ap.signalDbm > -65) TonalixColors.AccentEmerald else TonalixColors.TextSecondary,
                modifier = Modifier.size(18.dp)
            )
            Spacer(modifier = Modifier.width(10.dp))
            Column {
                Row(verticalAlignment = Alignment.CenterVertically) {
                    Text(
                        text = ap.ssid,
                        color = TonalixColors.TextPrimary,
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Bold
                    )
                    if (ap.isLocked) {
                        Spacer(modifier = Modifier.width(6.dp))
                        Icon(imageVector = Icons.Default.Lock, contentDescription = null, tint = TonalixColors.TextMuted, modifier = Modifier.size(11.dp))
                    }
                    if (ap.isSaved) {
                        Spacer(modifier = Modifier.width(6.dp))
                        Text(
                            text = "Guardada",
                            color = TonalixColors.PrimaryCyan,
                            fontSize = 8.sp,
                            fontFamily = FontFamily.Monospace,
                            modifier = Modifier.background(TonalixColors.PrimaryCyanGlow, RoundedCornerShape(3.dp)).padding(horizontal = 4.dp, vertical = 1.dp)
                        )
                    }
                }
                Spacer(modifier = Modifier.height(2.dp))
                Text(
                    text = "${ap.band.label} • ${ap.security.label} • ${ap.signalDbm} dBm • Máx ${ap.speedMbps} Mbps",
                    color = TonalixColors.TextMuted,
                    fontSize = 9.sp,
                    fontFamily = FontFamily.Monospace
                )
            }
        }

        Button(
            onClick = onConnect,
            enabled = !isConnectingThis,
            colors = ButtonDefaults.buttonColors(containerColor = if (isFocused) TonalixColors.PrimaryCyan else Color(0xFF131A22)),
            shape = RoundedCornerShape(6.dp),
            contentPadding = PaddingValues(horizontal = 12.dp, vertical = 6.dp)
        ) {
            if (isConnectingThis) {
                CircularProgressIndicator(modifier = Modifier.size(12.dp), color = TonalixColors.PrimaryCyan, strokeWidth = 2.dp)
                Spacer(modifier = Modifier.width(6.dp))
                Text(text = "Conectando...", color = TonalixColors.TextPrimary, fontSize = 10.sp)
            } else {
                Text(
                    text = "Conectar",
                    color = if (isFocused) Color.Black else TonalixColors.TextPrimary,
                    fontSize = 10.sp,
                    fontWeight = FontWeight.Bold
                )
            }
        }
    }
}
