// ============================================================================
/**
 * Punto de entrada con inyección de ViewModel y recolección con ciclo de vida consciente.
 * Diseñada para Android TV / Compose TV y 10-foot UI.
 */
@Composable
fun TonalixPerformanceRoute(
    viewModel: TonalixPerformanceViewModel = viewModel(),
    onNavigateBack: () -> Unit = {}
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    // Consumo de eventos puntuales (One-Shot UI Effects: Toasts, Haptics/Audio, Navegación)
    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is PerformanceUiEffect.ShowToast -> {
                    Toast.makeText(context, effect.message, Toast.LENGTH_SHORT).show()
                }
                is PerformanceUiEffect.PlaySoundEffect -> {
                    // Integración con AudioManager / View.playSoundEffect de Android TV
                }
                is PerformanceUiEffect.NavigateBack -> {
                    onNavigateBack()
                }
            }
        }
    }

    TonalixPerformanceScreen(
        uiState = uiState,
        onAction = viewModel::onAction,
        onNavigateBack = onNavigateBack
    )
}

/**
 * Versión pura / desacoplada (Stateless) para testeo y vista previa directa en Compose.
 */
@Composable
fun TonalixPerformanceScreen(
    uiState: PerformanceUiState,
    onAction: (PerformanceUserAction) -> Unit,
    onNavigateBack: () -> Unit = {}
) {
    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(TonalixColors.Background)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            // Barra superior fija con estado del sistema y telemetría en vivo
            TonalixTopBar(telemetry = uiState.telemetry)

            // Cuerpo principal: Sidebar Izquierdo + Contenido Central + Panel Derecho de Telemetría
            Row(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
            ) {
                // 1. Sidebar de navegación lateral
                TonalixSettingsSidebar(
                    selectedCategory = uiState.selectedCategory,
                    onSelectCategory = { onAction(PerformanceUserAction.SelectCategory(it)) }
                )

                // Separador vertical sutil
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 2. Contenido Central (Ajustes de Rendimiento y Memoria)
                Box(
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                ) {
                    LazyColumn(
                        modifier = Modifier
                            .fillMaxSize()
                            .padding(horizontal = 26.dp, vertical = 20.dp),
                        verticalArrangement = Arrangement.spacedBy(18.dp)
                    ) {
                        // Encabezado y acciones principales (Liberar RAM Boost y Aplicar)
                        item {
                            PerformanceHeaderSection(
                                activeProfileName = uiState.activeProfileName,
                                zramEnabled = uiState.zramDynamicSwapping,
                                isBoosting = uiState.isBoostingRam,
                                isApplying = uiState.isApplyingProfile,
                                onFreeRamBoost = { onAction(PerformanceUserAction.TriggerRamBoost) },
                                onApplyProfile = { onAction(PerformanceUserAction.ApplyCurrentProfile) }
                            )
                        }

                        // Tarjeta 1: Gobernador CPU & Frecuencia SoC (D-PAD Navegable)
                        item {
                            CpuGovernorSection(
                                selectedGovernor = uiState.selectedGovernor,
                                onSelectGovernor = { onAction(PerformanceUserAction.SelectGovernor(it)) }
                            )
                        }

                        // Tarjeta 2: Gestión de Memoria y Low-RAM Killer
                        item {
                            MemoryManagementSection(
                                autoKill = uiState.autoKillBackgroundApps,
                                onToggleAutoKill = { onAction(PerformanceUserAction.ToggleAutoKill(it)) },
                                selectedLimit = uiState.selectedProcessLimit,
                                onSelectLimit = { onAction(PerformanceUserAction.SelectProcessLimit(it)) },
                                zramEnabled = uiState.zramDynamicSwapping,
                                onToggleZram = { onAction(PerformanceUserAction.ToggleZram(it)) },
                                bufferPurge = uiState.bufferPurgeOnSuspend,
                                onToggleBufferPurge = { onAction(PerformanceUserAction.ToggleBufferPurge(it)) },
                                freePercent = uiState.telemetry.ramFreePercent
                            )
                        }

                        // Tarjeta 3: Renderizado Gráfico & 10-Foot UI Lean Mode
                        item {
                            GraphicRenderSection(
                                zeroAnimation = uiState.zeroAnimationScale,
                                onToggleZeroAnimation = { onAction(PerformanceUserAction.ToggleZeroAnimation(it)) },
                                forceGpu = uiState.forceGpuRendering,
                                onToggleForceGpu = { onAction(PerformanceUserAction.ToggleForceGpu(it)) },
                                selectedResolution = uiState.selectedResolution,
                                onSelectResolution = { onAction(PerformanceUserAction.SelectFramebufferResolution(it)) }
                            )
                        }
                    }
                }

                // Separador vertical
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 3. Panel Lateral Derecho (Telemetría SoC en Vivo 120 FPS y Ficha Técnica)
                TonalixPerformanceDiagnosticsSidebar(
                    telemetry = uiState.telemetry,
                    isRunningStressTest = uiState.isRunningStressTest,
                    stressProgress = uiState.stressTestProgress,
                    onStartStressTest = { onAction(PerformanceUserAction.StartStressTest) },
                    onCancelStressTest = { onAction(PerformanceUserAction.CancelStressTest) }
                )
            }

            // Barra inferior con guía de botones de control remoto D-PAD
            TonalixBottomBar()
        }
    }
}

@Composable
private fun SidebarCategoryItem(
    category: SettingsCategory,
    isSelected: Boolean,
    onClick: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val bgColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyanGlow
            isFocused -> TonalixColors.SurfaceCardHover
            else -> Color.Transparent
        }, label = "bg"
    )

    val borderColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyan
            isFocused -> TonalixColors.TextSecondary
            else -> Color.Transparent
        }, label = "border"
    )

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(1.dp, borderColor, RoundedCornerShape(8.dp))
            .focusable(interactionSource = interactionSource)
            .clickable(onClick = onClick)
            .padding(horizontal = 12.dp, vertical = 10.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Icon(
            imageVector = category.icon,
            contentDescription = null,
            tint = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextSecondary,
            modifier = Modifier.size(18.dp)
        )
        Spacer(modifier = Modifier.width(10.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = category.title,
                color = if (isSelected) TonalixColors.TextPrimary else TonalixColors.TextSecondary,
                fontSize = 12.sp,
                fontWeight = if (isSelected) FontWeight.SemiBold else FontWeight.Normal
            )
            Text(
                text = category.subtitle,
                color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextMuted,
                fontSize = 10.sp,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
        }
        if (isSelected) {
            Box(
                modifier = Modifier
                    .size(6.dp)
                    .clip(CircleShape)
                    .background(TonalixColors.PrimaryCyan)
            )
        }
    }
}

@Composable
private fun CpuGovernorCard(
    governor: CpuGovernor,
    isSelected: Boolean,
    onClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val borderColor = when {
        isSelected -> TonalixColors.PrimaryCyan
        isFocused -> TonalixColors.TextSecondary
        else -> TonalixColors.SurfaceBorder
    }

    val bgColor = when {
        isSelected -> Color(0xFF00222B)
        isFocused -> TonalixColors.SurfaceCardHover
        else -> Color(0xFF0A0F14)
    }

    Box(
        modifier = modifier
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(1.dp, borderColor, RoundedCornerShape(8.dp))
            .focusable(interactionSource = interactionSource)
            .clickable(onClick = onClick)
            .padding(12.dp)
    ) {
        Column {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Row(verticalAlignment = Alignment.CenterVertically) {
                    Text(
                        text = governor.title,
                        color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextPrimary,
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Bold
                    )
                    if (isSelected) {
                        Spacer(modifier = Modifier.width(6.dp))
                        Icon(
                            imageVector = Icons.Default.CheckCircle,
                            contentDescription = null,
                            tint = TonalixColors.PrimaryCyan,
                            modifier = Modifier.size(13.dp)
                        )
                    }
                }
                Text(
                    text = governor.speed,
                    color = if (isSelected) TonalixColors.AccentEmerald else TonalixColors.TextSecondary,
                    fontSize = 10.sp,
                    fontFamily = FontFamily.Monospace,
                    fontWeight = FontWeight.Bold
                )
            }

            Spacer(modifier = Modifier.height(4.dp))
            Text(
                text = governor.description,
                color = TonalixColors.TextSecondary,
                fontSize = 10.sp,
                lineHeight = 13.sp
            )

            Spacer(modifier = Modifier.height(8.dp))
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = governor.tag,
                    color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextMuted,
                    fontSize = 8.sp,
                    fontFamily = FontFamily.Monospace,
                    fontWeight = FontWeight.Bold,
                    modifier = Modifier
                        .background(if (isSelected) TonalixColors.PrimaryCyanGlow else Color(0xFF0E141A), RoundedCornerShape(3.dp))
                        .padding(horizontal = 4.dp, vertical = 2.dp)

                    Text(text = "GPU Boost Mali +30%", color = TonalixColors.AccentEmerald, fontSize = 8.sp, fontFamily = FontFamily.Monospace)
                }

        }
    }
