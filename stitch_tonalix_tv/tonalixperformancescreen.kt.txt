/**
 * Pantalla de Modo Rendimiento (Optimización Ultra-Lite) para Tonalix TV OS.
 * Optimizada para control remoto D-Pad 10-foot UI en Smart TV / Android TV Boxes.
 */
@Composable
fun TonalixPerformanceScreen(
    onNavigateBack: () -> Unit = {},
    onFreeRamBoost: () -> Unit = {},
    onApplyProfile: () -> Unit = {},
    onRunStressTest: () -> Unit = {}
) {
    var selectedCategory by remember { mutableStateOf(SettingsCategory.PERFORMANCE) }
    var selectedGovernor by remember { mutableStateOf(CpuGovernor.GAMER_120FPS) }
    var selectedProcessLimit by remember { mutableStateOf(BackgroundProcessLimit.ONE_PROCESS) }
    var selectedResolution by remember { mutableStateOf(FramebufferResolution.RES_1080P) }

    // Toggles de Optimización
    var autoKillBackgroundApps by remember { mutableStateOf(true) }
    var zramDynamicSwapping by remember { mutableStateOf(true) }
    var bufferPurgeOnSuspend by remember { mutableStateOf(true) }
    var zeroAnimationScale by remember { mutableStateOf(true) }
    var forceGpuRendering by remember { mutableStateOf(true) }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(TonalixColors.Background)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            // Barra superior fija con estado del sistema y telemetría
            TonalixTopBar()

            // Cuerpo principal: Sidebar Izquierdo + Contenido Central + Panel Derecho de Telemetría
            Row(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
            ) {
                // 1. Sidebar de navegación lateral
                TonalixSettingsSidebar(
                    selectedCategory = selectedCategory,
                    onSelectCategory = { selectedCategory = it }
                )

                // Separador vertical sutil
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 2. Contenido Central (Ajustes de Rendimiento y Memoria)
                Box(
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                ) {
                    LazyColumn(
                        modifier = Modifier
                            .fillMaxSize()
                            .padding(horizontal = 26.dp, vertical = 20.dp),
                        verticalArrangement = Arrangement.spacedBy(18.dp)
                    ) {
                        // Encabezado y acciones principales (Liberar RAM Boost y Aplicar)
                        item {
                            PerformanceHeaderSection(
                                onFreeRamBoost = onFreeRamBoost,
                                onApplyProfile = onApplyProfile
                            )
                        }

                        // Tarjeta 1: Gobernador CPU & Frecuencia SoC (D-PAD Navegable)
                        item {
                            CpuGovernorSection(
                                selectedGovernor = selectedGovernor,
                                onSelectGovernor = { selectedGovernor = it }
                            )
                        }

                        // Tarjeta 2: Gestión de Memoria y Low-RAM Killer
                        item {
                            MemoryManagementSection(
                                autoKill = autoKillBackgroundApps,
                                onToggleAutoKill = { autoKillBackgroundApps = it },
                                selectedLimit = selectedProcessLimit,
                                onSelectLimit = { selectedProcessLimit = it },
                                zramEnabled = zramDynamicSwapping,
                                onToggleZram = { zramDynamicSwapping = it },
                                bufferPurge = bufferPurgeOnSuspend,
                                onToggleBufferPurge = { bufferPurgeOnSuspend = it }
                            )
                        }

                        // Tarjeta 3: Renderizado Gráfico & 10-Foot UI Lean Mode
                        item {
                            GraphicRenderSection(
                                zeroAnimation = zeroAnimationScale,
                                onToggleZeroAnimation = { zeroAnimationScale = it },
                                forceGpu = forceGpuRendering,
                                onToggleForceGpu = { forceGpuRendering = it },
                                selectedResolution = selectedResolution,
                                onSelectResolution = { selectedResolution = it }
                            )
                        }
                    }
                }

                // Separador vertical
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 3. Panel Lateral Derecho (Telemetría SoC en Vivo 120 FPS y Ficha Técnica)
                TonalixPerformanceDiagnosticsSidebar(
                    onRunStressTest = onRunStressTest
                )
            }

            // Barra inferior con guía de botones de control remoto D-PAD
            TonalixBottomBar()
        }
    }
}

@Composable
private fun SidebarCategoryItem(
    category: SettingsCategory,
    isSelected: Boolean,
    onClick: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val bgColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyanGlow
            isFocused -> TonalixColors.SurfaceCardHover
            else -> Color.Transparent
        }, label = "bg"
    )

    val borderColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyan
            isFocused -> TonalixColors.TextSecondary
            else -> Color.Transparent
        }, label = "border"
    )

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(1.dp, borderColor, RoundedCornerShape(8.dp))
            .focusable(interactionSource = interactionSource)
            .clickable(onClick = onClick)
            .padding(horizontal = 12.dp, vertical = 10.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Icon(
            imageVector = category.icon,
            contentDescription = null,
            tint = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextSecondary,
            modifier = Modifier.size(18.dp)
        )
        Spacer(modifier = Modifier.width(10.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = category.title,
                color = if (isSelected) TonalixColors.TextPrimary else TonalixColors.TextSecondary,
                fontSize = 12.sp,
                fontWeight = if (isSelected) FontWeight.SemiBold else FontWeight.Normal
            )
            Text(
                text = category.subtitle,
                color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextMuted,
                fontSize = 10.sp,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
        }
        if (isSelected) {
            Box(
                modifier = Modifier
                    .size(6.dp)
                    .clip(CircleShape)
                    .background(TonalixColors.PrimaryCyan)
            )
        }
    }
}

    Row(
        modifier = Modifier.fillMaxWidth(),
        horizontalArrangement = Arrangement.SpaceBetween,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(text = label, color = TonalixColors.TextMuted, fontSize = 10.sp)
        Text(text = value, color = TonalixColors.TextPrimary, fontSize = 10.sp, fontFamily = FontFamily.Monospace)
    }
}

@Composable
private fun TonalixBottomBar() {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .height(38.dp)
            .background(TonalixColors.Surface)
            .border(width = 1.dp, color = TonalixColors.SurfaceBorder)
            .padding(horizontal = 24.dp),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.SpaceBetween
    ) {
        Row(
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(16.dp)
        ) {
            DpadKeyGuide(key = "▲▼", label = "Navegar")

                color = TonalixColors.TextMuted,
                fontSize = 11.sp,
                fontFamily = FontFamily.Monospace

@Composable
private fun DpadKeyGuide(key: String, label: String) {
    Row(verticalAlignment = Alignment.CenterVertically) {
        Box(
            modifier = Modifier
                .clip(RoundedCornerShape(4.dp))
                .background(Color(0xFF1B2430))
                .border(1.dp, TonalixColors.SurfaceBorder, RoundedCornerShape(4.dp))
                .padding(horizontal = 5.dp, vertical = 2.dp)
        ) {
            Text(
                text = key,
                color = TonalixColors.TextPrimary,
                fontSize = 9.sp,
                fontWeight = FontWeight.Bold,
                fontFamily = FontFamily.Monospace
            )
        }
        Spacer(modifier = Modifier.width(6.dp))
        Text(
            text = label,
            color = TonalixColors.TextSecondary,
            fontSize = 11.sp
        )
    }
}
