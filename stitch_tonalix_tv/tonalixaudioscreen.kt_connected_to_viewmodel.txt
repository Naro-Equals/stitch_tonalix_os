/**
 * Punto de entrada oficial de Ajustes de Audio con inyección de ViewModel y
 * recolección con ciclo de vida consciente para Android TV / Compose TV.
 */
@Composable
fun TonalixAudioRoute(
    viewModel: TonalixAudioViewModel = viewModel(),
    onNavigateBack: () -> Unit = {}
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    // Consumo de eventos puntuales (Side Effects / Haptic Feedback / Toasts)
    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is AudioUiEffect.ShowToast -> {
                    Toast.makeText(context, effect.message, Toast.LENGTH_SHORT).show()
                }
                is AudioUiEffect.PlayHapticTone -> {
                    // Integración con SoundPool / View.playSoundEffect del control remoto
                }
                is AudioUiEffect.NavigateBack -> {
                    onNavigateBack()
                }
            }
        }
    }

    TonalixAudioScreen(
        uiState = uiState,
        onAction = viewModel::onAction,
        onNavigateBack = onNavigateBack
    )
}

/**
 * Vista pura y desacoplada (Stateless) para pruebas unitarias y previews en Android Studio.
 */
@Composable
fun TonalixAudioScreen(
    uiState: AudioUiState,
    onAction: (AudioUserAction) -> Unit,
    onNavigateBack: () -> Unit = {}
) {
    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(TonalixColors.Background)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            // Barra superior fija con estado del sistema y telemetría de audio
            TonalixTopBar(telemetry = uiState.telemetry)

            // Cuerpo principal: Sidebar Izquierdo + Contenido Central + Sidebar Derecho (Telemetría DSP)
            Row(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
            ) {
                // 1. Sidebar de navegación lateral
                TonalixSettingsSidebar(
                    selectedCategory = SettingsCategory.AUDIO,
                    onSelectCategory = { /* Navegación entre categorías */ }
                )

                // Separador vertical sutil
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 2. Contenido Central (Ajustes de Audio, eARC y Calibración Acústica)
                Box(
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                ) {
                    LazyColumn(
                        modifier = Modifier
                            .fillMaxSize()
                            .padding(horizontal = 26.dp, vertical = 20.dp),
                        verticalArrangement = Arrangement.spacedBy(18.dp)
                    ) {
                        // Encabezado y acciones principales (Restablecer EDID y Aplicar)
                        item {
                            AudioHeaderSection(
                                isApplying = uiState.isApplyingChanges,
                                onResetEdid = { onAction(AudioUserAction.ResetAudioEdid) },
                                onApplySettings = { onAction(AudioUserAction.ApplyAudioSettings) }
                            )
                        }

                        // Banner de Decodificación y Handshake HDMI 2.1 eARC
                        item {
                            AudioDetectionBanner(telemetry = uiState.telemetry)
                        }

                        // Tarjeta 1: Modos de Salida de Audio Primaria (HDMI eARC, Óptico, BT, TV)
                        item {
                            AudioOutputModesSection(
                                selectedMode = uiState.selectedOutputMode,
                                onSelectMode = { onAction(AudioUserAction.SelectOutputMode(it)) }
                            )
                        }

                        // Tarjeta 2: Formatos Surround, Spatial Audio Dolby Atmos & DTS:X
                        item {
                            SpatialAudioFormatsCard(
                                dolbyAtmos = uiState.dolbyAtmosSpatialAudio,
                                onToggleDolbyAtmos = { onAction(AudioUserAction.ToggleDolbyAtmos(it)) },
                                dtsX = uiState.dtsXBitstreamPassthrough,
                                onToggleDtsX = { onAction(AudioUserAction.ToggleDtsX(it)) },
                                losslessTrueHd = uiState.losslessAudioTrueHd,
                                onToggleLossless = { onAction(AudioUserAction.ToggleLosslessTrueHd(it)) },
                                selectedHandling = uiState.selectedFormatHandling,
                                onSelectHandling = { onAction(AudioUserAction.SelectFormatHandling(it)) }
                            )
                        }

                        // Tarjeta 3: Calibración Acústica de Sala & Micrófono Remoto
                        item {
                            RoomAcousticsCalibrationCard(
                                selectedProfile = uiState.selectedAcousticProfile,
                                onSelectProfile = { onAction(AudioUserAction.SelectAcousticProfile(it)) },
                                isCalibrating = uiState.isCalibratingRoom,
                                calibrationProgress = uiState.calibrationProgress,
                                confidence = uiState.roomCalibrationConfidence,
                                lastTimestamp = uiState.lastCalibrationTimestamp,
                                onStartCalibration = { onAction(AudioUserAction.StartRoomCalibration) },
                                onCancelCalibration = { onAction(AudioUserAction.CancelRoomCalibration) }
                            )
                        }

                        // Tarjeta 4: Mejoras Vocales, Sincronización A/V Lip-Sync y Rango Dinámico
                        item {
                            DialogueAndSyncCard(
                                vocalBoost = uiState.vocalBoostEnhancer,
                                onToggleVocalBoost = { onAction(AudioUserAction.ToggleVocalBoost(it)) },
                                clarityLevel = uiState.dialogueClarityLevel,
                                onSetClarity = { onAction(AudioUserAction.SetDialogueClarity(it)) },
                                avSyncMs = uiState.avSyncDelayMs,
                                onAdjustAvSync = { onAction(AudioUserAction.AdjustAvSyncDelay(it)) },
                                selectedDrc = uiState.selectedDrc,
                                onSelectDrc = { onAction(AudioUserAction.SelectDrc(it)) },
                                subBassDb = uiState.subBassBoostDb,
                                onSetBass = { onAction(AudioUserAction.SetBassBoost(it)) }
                            )
                        }
                    }
                }

                // Separador vertical
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 3. Panel Lateral Derecho (Diagnóstico DSP en Vivo, VU Meters y Prueba Rosa)
                TonalixAudioDiagnosticsSidebar(
                    telemetry = uiState.telemetry,
                    isTestingTone = uiState.isTestingPinkNoise,
                    onPlayTone = { onAction(AudioUserAction.PlayTestTone) },
                    onStopTone = { onAction(AudioUserAction.StopTestTone) }
                )
            }

            // Barra inferior con guía de botones de control remoto D-PAD
            TonalixBottomBar()
        }
    }
}

@Composable
private fun SidebarCategoryItem(
    category: SettingsCategory,
    isSelected: Boolean,
    onClick: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val bgColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyanGlow
            isFocused -> TonalixColors.SurfaceCardHover
            else -> Color.Transparent
        }, label = "bg"
    )

    val borderColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyan
            isFocused -> TonalixColors.TextSecondary
            else -> Color.Transparent
        }, label = "border"
    )

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(1.dp, borderColor, RoundedCornerShape(8.dp))
            .focusable(interactionSource = interactionSource)
            .clickable(onClick = onClick)
            .padding(horizontal = 12.dp, vertical = 10.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Icon(
            imageVector = category.icon,
            contentDescription = null,
            tint = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextSecondary,
            modifier = Modifier.size(18.dp)
        )
        Spacer(modifier = Modifier.width(10.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = category.title,
                color = if (isSelected) TonalixColors.TextPrimary else TonalixColors.TextSecondary,
                fontSize = 12.sp,
                fontWeight = if (isSelected) FontWeight.SemiBold else FontWeight.Normal
            )
            Text(
                text = category.subtitle,
                color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextMuted,
                fontSize = 10.sp,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
        }
        if (isSelected) {
            Box(
                modifier = Modifier
                    .size(6.dp)
                    .clip(CircleShape)
                    .background(TonalixColors.PrimaryCyan)
            )
        }
    }
}

                        Text(text = "Atenúa estruendos para escuchar a bajo volumen nocturno.", color = TonalixColors.TextMuted, fontSize = 9.sp, lineHeight = 12.sp)
                        Spacer(modifier = Modifier.height(8.dp))
                        DynamicRangeCompression.values().forEach { drc ->
                            val isSelected = drc == selectedDrc
                            Row(
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .clickable { onSelectDrc(drc) }
                                    .padding(vertical = 2.dp),
                                verticalAlignment = Alignment.CenterVertically
                            ) {
                                Box(
                                    modifier = Modifier
                                        .size(10.dp)
                                        .clip(CircleShape)
                                        .background(if (isSelected) TonalixColors.PrimaryCyan else Color(0xFF222D39))
                                )
                                Spacer(modifier = Modifier.width(6.dp))
                                Text(
                                    text = drc.label,
                                    color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextSecondary,
                                    fontSize = 10.sp,
                                    fontWeight = if (isSelected) FontWeight.Bold else FontWeight.Normal
                                )
                            }
                        }
                    }
                }
            }
        }
    }
