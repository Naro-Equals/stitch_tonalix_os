/**
 * Punto de entrada oficial de Privacidad y Seguridad con inyección de ViewModel y
 * recolección con ciclo de vida consciente para Android TV / Compose TV.
 */
@Composable
fun TonalixPrivacyRoute(
    viewModel: TonalixPrivacyViewModel = viewModel(),
    onNavigateBack: () -> Unit = {}
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    // Consumo de eventos puntuales (Side Effects / Haptic Feedback / Toasts)
    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is PrivacyUiEffect.ShowToast -> {
                    Toast.makeText(context, effect.message, Toast.LENGTH_SHORT).show()
                }
                is PrivacyUiEffect.PlaySound -> {
                    // Integración con SoundPool / View.playSoundEffect del control remoto
                }
                is PrivacyUiEffect.NavigateBack -> {
                    onNavigateBack()
                }
            }
        }
    }

    TonalixPrivacyScreen(
        uiState = uiState,
        onAction = viewModel::onAction,
        onNavigateBack = onNavigateBack
    )
}

/**
 * Vista pura y desacoplada (Stateless) para pruebas unitarias y previews en Android Studio.
 */
@Composable
fun TonalixPrivacyScreen(
    uiState: PrivacyUiState,
    onAction: (PrivacyUserAction) -> Unit,
    onNavigateBack: () -> Unit = {}
) {
    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(TonalixColors.Background)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            // Barra superior fija con estado del sistema y telemetría de seguridad
            TonalixTopBar(telemetry = uiState.telemetry)

            // Cuerpo principal: Sidebar Izquierdo + Contenido Central + Sidebar Derecho (Telemetría de Red y DNS)
            Row(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
            ) {
                // 1. Sidebar de navegación lateral
                TonalixSettingsSidebar(
                    selectedCategory = SettingsCategory.PRIVACY,
                    onSelectCategory = { /* Navegación entre categorías */ }
                )

                // Separador vertical sutil
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 2. Contenido Central (Ajustes de Privacidad, DNS Seguro y Filtro Anti-Ads)
                Box(
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                ) {
                    LazyColumn(
                        modifier = Modifier
                            .fillMaxSize()
                            .padding(horizontal = 26.dp, vertical = 20.dp),
                        verticalArrangement = Arrangement.spacedBy(18.dp)
                    ) {
                        // Encabezado y acciones principales (Purgar Caché DNS y Aplicar)
                        item {
                            PrivacyHeaderSection(
                                isApplying = uiState.isApplyingSettings,
                                onFlushCache = { onAction(PrivacyUserAction.FlushDnsCache) },
                                onApplySettings = { onAction(PrivacyUserAction.ApplyPrivacySettings) }
                            )
                        }

                        // Banner de Telemetría Activa de Bloqueo y Estado del Cortafuegos
                        item {
                            SecurityDetectionBanner(telemetry = uiState.telemetry, totalBlocked = uiState.totalBlockedQueries)
                        }

                        // Tarjeta 1: Servidores DNS Seguros Cifrados (DoH / DoT)
                        item {
                            DnsProvidersSection(
                                selectedProvider = uiState.selectedDnsProvider,
                                onSelectProvider = { onAction(PrivacyUserAction.SelectDnsProvider(it)) }
                            )
                        }

                        // Tarjeta 2: Escudo Anti-Publicidad Global & Anti-Rastreadores OEM
                        item {
                            AdBlockAndTelemetryShieldCard(
                                adBlockGlobal = uiState.adBlockGlobalEnabled,
                                onToggleAdBlock = { onAction(PrivacyUserAction.ToggleAdBlockGlobal(it)) },
                                telemetryOem = uiState.telemetryBlockOemEnabled,
                                onToggleTelemetryOem = { onAction(PrivacyUserAction.ToggleTelemetryBlockOem(it)) },
                                smartTubeShield = uiState.smartTubeAdFreeShield,
                                onToggleSmartTube = { onAction(PrivacyUserAction.ToggleSmartTubeShield(it)) },
                                selectedLevel = uiState.selectedBlockingLevel,
                                onSelectLevel = { onAction(PrivacyUserAction.SelectBlockingLevel(it)) },
                                onUpdateRules = { onAction(PrivacyUserAction.UpdateBlocklists) },
                                isUpdatingRules = uiState.isUpdatingBlocklists,
                                activeRulesCount = uiState.telemetry.activeRulesCount
                            )
                        }

                        // Tarjeta 3: Sesión Blindada, Anonimización y Fugas DNS
                        item {
                            ArmoredSessionAndLeaksCard(
                                isolateSession = uiState.isolateArmoredSession,
                                onToggleIsolate = { onAction(PrivacyUserAction.ToggleArmoredSession(it)) },
                                preventLeak = uiState.preventDnsLeak,
                                onTogglePreventLeak = { onAction(PrivacyUserAction.TogglePreventDnsLeak(it)) },
                                anonymizeMac = uiState.anonymizeMacAddress,
                                onToggleAnonymizeMac = { onAction(PrivacyUserAction.ToggleMacAnonymization(it)) },
                                selectedEncryption = uiState.selectedEncryption,
                                onSelectEncryption = { onAction(PrivacyUserAction.SelectEncryption(it)) }
                            )
                        }
                    }
                }

                // Separador vertical
                Box(
                    modifier = Modifier
                        .fillMaxHeight()
                        .width(1.dp)
                        .background(TonalixColors.SurfaceBorder)
                )

                // 3. Panel Lateral Derecho (Diagnóstico en Vivo, Ping DNS, Test de Fugas)
                TonalixPrivacyDiagnosticsSidebar(
                    telemetry = uiState.telemetry,
                    isTestingLeak = uiState.isTestingDnsLeak,
                    onRunLeakTest = { onAction(PrivacyUserAction.RunDnsLeakTest) },
                    onResetDefaults = { onAction(PrivacyUserAction.ResetPrivacyDefaults) }
                )
            }

            // Barra inferior con guía de botones de control remoto D-PAD
            TonalixBottomBar()
        }
    }
}

@Composable
private fun SidebarCategoryItem(
    category: SettingsCategory,
    isSelected: Boolean,
    onClick: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val bgColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyanGlow
            isFocused -> TonalixColors.SurfaceCardHover
            else -> Color.Transparent
        }, label = "bg"
    )

    val borderColor by animateColorAsState(
        targetValue = when {
            isSelected -> TonalixColors.PrimaryCyan
            isFocused -> TonalixColors.TextSecondary
            else -> Color.Transparent
        }, label = "border"
    )

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(1.dp, borderColor, RoundedCornerShape(8.dp))
            .focusable(interactionSource = interactionSource)
            .clickable(onClick = onClick)
            .padding(horizontal = 12.dp, vertical = 10.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Icon(
            imageVector = category.icon,
            contentDescription = null,
            tint = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextSecondary,
            modifier = Modifier.size(18.dp)
        )
        Spacer(modifier = Modifier.width(10.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = category.title,
                color = if (isSelected) TonalixColors.TextPrimary else TonalixColors.TextSecondary,
                fontSize = 12.sp,
                fontWeight = if (isSelected) FontWeight.SemiBold else FontWeight.Normal
            )
            Text(
                text = category.subtitle,
                color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextMuted,
                fontSize = 10.sp,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
        }
        if (isSelected) {
            Box(
                modifier = Modifier
                    .size(6.dp)
                    .clip(CircleShape)
                    .background(TonalixColors.PrimaryCyan)
            )
        }
    }
}

@Composable
private fun DnsProviderCard(
    provider: DnsProvider,
    isSelected: Boolean,
    onClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()

    val borderColor = when {
        isSelected -> TonalixColors.PrimaryCyan
        isFocused -> TonalixColors.TextSecondary
        else -> TonalixColors.SurfaceBorder
    }

    val bgColor = when {
        isSelected -> Color(0xFF00222B)
        isFocused -> TonalixColors.SurfaceCardHover
        else -> Color(0xFF0A0F14)
    }

    Box(
        modifier = modifier
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(1.dp, borderColor, RoundedCornerShape(8.dp))
            .focusable(interactionSource = interactionSource)
            .clickable(onClick = onClick)
            .padding(12.dp)
    ) {
        Column {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Row(verticalAlignment = Alignment.CenterVertically) {
                    Text(
                        text = provider.title,
                        color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextPrimary,
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Bold
                    )
                    if (isSelected) {
                        Spacer(modifier = Modifier.width(6.dp))
                        Icon(
                            imageVector = Icons.Default.CheckCircle,
                            contentDescription = null,
                            tint = TonalixColors.PrimaryCyan,
                            modifier = Modifier.size(13.dp)
                        )
                    }
                }
                Text(
                    text = provider.tag,
                    color = if (isSelected) TonalixColors.PrimaryCyan else TonalixColors.TextMuted,
                    fontSize = 8.sp,
                    fontFamily = FontFamily.Monospace,
                    fontWeight = FontWeight.Bold,
                    modifier = Modifier
                        .background(if (isSelected) TonalixColors.PrimaryCyanGlow else Color(0xFF0E141A), RoundedCornerShape(3.dp))
                        .padding(horizontal = 4.dp, vertical = 2.dp)
                )
            }

            Spacer(modifier = Modifier.height(4.dp))
            Text(
                text = "${provider.ipPrimary} • ${provider.ipSecondary} (${provider.protocol})",
                color = TonalixColors.TextSecondary,
                fontSize = 10.sp,
                fontFamily = FontFamily.Monospace
            )
        }
    }
